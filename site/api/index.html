baselines ¶

Step 2 of RQ2: baseline forecasts every later model has to beat.

Baselines (all use only information available at the forecast origin): persistence forecast = last value rolling_mean_k forecast = mean of the last k values segment_mean forecast = mean of the indicator since the segment started

Scores: MAE and RMSE (in the indicator's unit, m/s), and skill versus persistence (1 - MAE / MAE_persistence; 0 = no better than persistence, positive = better, negative = worse).

Usage

ev = BaselineEvaluation(p, horizons=(1, 5, 15), lookback=10).run() ev.report() ev.save("results")

Shell (on a prepared csv written by prepare_oo.py): python baselines.py

class BaselineEvaluation:
    def __init__(self, source, targets=INDICATORS, horizons=(1, 5, 15), lookback=10,
                 baselines=None, train=0.6, val=0.2, **dataset_kw):
        self.source = source
        self.targets = targets
        self.horizons = horizons
        self.lookback = lookback
        self.baselines = baselines or default_baselines(lookback)
        self.train, self.val = train, val
        self.dataset_kw = dataset_kw
        self.datasets = {}          # (target, horizon) -> ForecastDataset, reused by later models
        self.results = pd.DataFrame()

    def run(self):
        rows = []
        for target in self.targets:
            for h in self.horizons:
                ds = ForecastDataset(self.source, target, horizon=h, lookback=self.lookback,
                                     **self.dataset_kw).check()
                try:
                    splits = ds.split(self.train, self.val)
                except ValueError as e:
                    print(f"  skipped: {e}")
                    continue
                self.datasets[(target, h)] = ds
                for b in self.baselines:
                    b.fit(splits["train"])
                    for name in ("val", "test"):
                        s = splits[name]
                        pred = b.predict(s.X)
                        rows.append({"target": target, "horizon": h, "split": name,
                                     "baseline": b.name, "n": len(s),
                                     "mae": mae(s.y, pred), "rmse": rmse(s.y, pred)})
        res = pd.DataFrame(rows)
        if not res.empty:
            ref = (res[res["baseline"] == "persistence"]
                   .set_index(["target", "horizon", "split"])["mae"].rename("mae_persistence"))
            res = res.join(ref, on=["target", "horizon", "split"])
            res["skill_vs_persistence"] = 1 - res["mae"] / res["mae_persistence"]
            res = res.drop(columns="mae_persistence")
        self.results = res
        return self

    def table(self, split="test", metric="mae"):
        """Rows = target and horizon, columns = baselines."""
        r = self.results[self.results["split"] == split]
        return r.pivot_table(index=["target", "horizon"], columns="baseline", values=metric)

    def best(self, split="test"):
        """Best baseline per target and horizon: the number later models must beat."""
        r = self.results[self.results["split"] == split]
        idx = r.groupby(["target", "horizon"])["mae"].idxmin()
        return r.loc[idx, ["target", "horizon", "baseline", "mae", "rmse", "skill_vs_persistence"]] \
                .set_index(["target", "horizon"])

    def report(self):
        pd.set_option("display.width", 200, "display.max_columns", 20, "display.float_format", "{:.4f}".format)
        if self.results.empty:
            print("no results")
            return self
        ds = next(iter(self.datasets.values()))
        n = {k: len(v) for k, v in ds.split(self.train, self.val).items()}
        print(f"\nBaselines: lookback {self.lookback} min, samples per split (first target/horizon): {n}")
        print("\nTest MAE (m/s):\n", self.table("test", "mae"))
        print("\nValidation MAE (m/s):\n", self.table("val", "mae"))
        print("\nBest baseline on test (the bar for XGBoost / LSTM):\n", self.best("test"))
        return self

    def save(self, out_dir):
        out = Path(out_dir)
        out.mkdir(parents=True, exist_ok=True)
        self.results.to_csv(out / "baseline_results.csv", index=False)
        print(f"  saved {out / 'baseline_results.csv'}")
        return self

forecast_data ¶

Step 1 of RQ2: turn a prepared run into a supervised forecasting dataset.

Task: given the last lookback minutes of one indicator plus the process inputs at the forecast origin, predict that indicator horizon minutes later.

Windows are built inside segments only (a segment = unbroken block of valid, flowing minutes), so no sample ever spans a zero-flow gap. Splits are chronological and keep whole segments together, so no segment is shared between train, validation and test.

Columns of X lag_0 ... lag_{L-1} indicator at the origin (lag_0) and the L-1 minutes before it seg_mean mean of the indicator from the segment start up to the origin since_start minutes since the segment's first used row q_vol, p, T, mw ... process inputs at the origin

Usage

ds = ForecastDataset(p, target="dev_p1", horizon=5, lookback=10) ds.check() splits = ds.split(train=0.6, val=0.2) # {"train": Split, "val": Split, "test": Split}

class ForecastDataset:
    def __init__(self, source, target, horizon=1, lookback=10, covariates=COVARIATES, trim_min=1):
        df = source.df if hasattr(source, "df") else source
        for col in ("valid", "segment", target):
            if col not in df.columns:
                raise ValueError(f"missing column {col!r}; run Prepare.prepare() first")
        self.full = df
        self.target = target
        self.horizon = horizon
        self.lookback = lookback
        self.covariates = [c for c in covariates if c in df.columns]
        self.trim_min = trim_min
        self.X, self.y, self.meta = self._build()

    # ---- construction --------------------------------------------------------------------
    def _build(self):
        L, h = self.lookback, self.horizon
        v = self.full[self.full["valid"]]
        v = v[v.groupby("segment").cumcount() >= self.trim_min]     # drop restart transient
        g = v.groupby("segment")[self.target]

        feats = {f"lag_{k}": g.shift(k) for k in range(L)}           # shift never crosses segments
        feats["seg_mean"] = g.expanding().mean().reset_index(level=0, drop=True)
        X = pd.DataFrame(feats, index=v.index)
        for c in self.covariates:
            X[c] = v[c]
        X["since_start"] = v.groupby("segment").cumcount()

        y = g.shift(-h)
        meta = pd.DataFrame({"segment": v["segment"], "t_origin": v.index,
                             "t_target": v.index + pd.Timedelta(minutes=h)}, index=v.index)
        ok = X.notna().all(axis=1) & y.notna()
        return X[ok], y[ok].rename("y"), meta[ok]

    def __len__(self):
        return len(self.y)

    # ---- safety check ----------------------------------------------------------------------
    def check(self):
        """Verify against the raw time series that no sample crosses a gap or a segment."""
        one = pd.Timedelta(minutes=1)
        full = self.full
        t0, t1 = self.meta["t_origin"], self.meta["t_target"]
        tl = t0 - (self.lookback - 1) * one
        assert full.loc[t1, "valid"].all(), "a target lies outside valid rows"
        assert full.loc[tl, "valid"].all(), "a lookback start lies outside valid rows"
        seg = self.meta["segment"].to_numpy()
        assert (full.loc[t1, "segment"].to_numpy() == seg).all(), "target in another segment"
        assert (full.loc[tl, "segment"].to_numpy() == seg).all(), "lookback in another segment"
        assert np.allclose(full.loc[t1, self.target].to_numpy(), self.y.to_numpy()), "target mismatch"
        assert np.allclose(full.loc[tl, self.target].to_numpy(),
                           self.X[f"lag_{self.lookback - 1}"].to_numpy()), "lag mismatch"
        return self

    # ---- chronological split by whole segments ------------------------------------------------
    def split(self, train=0.6, val=0.2):
        counts = self.meta.groupby("segment").size().sort_index()    # segment ids increase with time
        total = counts.sum()
        before = counts.cumsum() - counts
        mid = (before + counts / 2) / total                          # where in time the segment sits
        which = np.where(mid < train, "train", np.where(mid < train + val, "val", "test"))
        owner = pd.Series(which, index=counts.index)
        part = self.meta["segment"].map(owner)

        out = {}
        for name in ("train", "val", "test"):
            m = (part == name).to_numpy()
            if not m.any():
                raise ValueError(f"{self.target} h={self.horizon}: the {name} split is empty "
                                 f"({len(counts)} segments, {total} samples); need more data or smaller fractions")
            out[name] = Split(name, self.X[m], self.y[m], self.meta[m])
        return out

    def summary(self):
        return (f"{self.target} h={self.horizon} L={self.lookback}: {len(self)} samples "
                f"in {self.meta['segment'].nunique()} segments")

class USMLoader:
    """Reads the USM Excel export and builds one DataFrame per run sheet.

    Usage:
        loader = USMLoader("input/Vega 2st stage low flow.xlsx")
        meter_data, gas_data = loader.load()
    After load(), the results are also kept on the object as loader.runs and loader.gas.
    """
    def __init__(self, path=FILE):
        self.path = path
        self.meter_data = {}      # sheet name -> DataFrame, filled by load()
        self.gas_data = None     # gas-quality DataFrame, filled by load()

    @staticmethod
    def is_datetime_value(x) -> bool:
        """True if a single cell value is a date/time object."""
        return isinstance(x, datetime.datetime)

    @staticmethod
    def to_datetime(col):
        """Convert one timestamp column from the Excel sheet into proper pandas datetimes.
        The column has dtype "object", but some cells are datetimes or
        when the column also contains text or blanks.
        The result is rounded to whole seconds, because Excel's
        floating-point storage gives values like 12:00:00.000001 that would
        otherwise fail to match when joining sheets on time.
        """
        is_dt = col.map(USMLoader.is_datetime_value)
        only_dates = col.where(is_dt)
        return pd.to_datetime(only_dates, errors="coerce").dt.round("s")

    def read_sheet(self, sheet, rename):
        """ 
        Takes one excel sheet, renames columns.

        """
        raw = pd.read_excel(self.path, sheet_name=sheet, header=None)

        names = raw.iloc[HEADER_ROW].tolist()
        df = raw.iloc[DATA_ROW:].copy()

        time_columns = ["t_start", "t_end"]

        value_columns = []
        for original_name in names[2:]: 
            if original_name in rename:
                new_name = rename[original_name]
            else:
                new_name = original_name
            value_columns.append(new_name)

        # Put the two lists together and use them as the column names
        df.columns = time_columns + value_columns
        #convert to real time
        df["t_start"] = self.to_datetime(df["t_start"])
        #drop missing t values and not need end time
        df = df.dropna(subset=["t_start"]).drop(columns="t_end")

        #  Keep only the columns that have a name
        named_columns = []
        for column_names in df.columns:
            if not pd.isna(column_names):
                named_columns.append(column_names)

        # index first, so it is not coerced to numbers
        df = df[named_columns]
        df = df.set_index("t_start").sort_index()  
        # making every value a number, coerce makes bad data to nan   
        return df.apply(pd.to_numeric, errors="coerce")

    @staticmethod
    def add_indicators(df):
        """cross-path VOS deviation: path minus mean of the other paths

            input parameters:
            df : dtype        
        """
        for path in PATHS:         
            # (a) find the names of the OTHER two paths
            other_paths = []
            for candidate in PATHS:
                if candidate != path:
                    other_paths.append(candidate)

            # (b) this path's velocity of sound: f"vos_{path}" becomes "vos_p1"
            this_path_vos = df[f"vos_{path}"]   

            # (c) the other two paths' columns, and their average for each minute
            other_column_names = []
            for other in other_paths:
                other_column_names.append(f"vos_{other}")
            other_paths_vos = df[other_column_names]
            # axis=1 means: average across the row
            average_of_others = other_paths_vos.mean(axis=1)   

            # (d) the new column: this path minus the average of the others
            df[f"dev_{path}"] = this_path_vos - average_of_others

        # GC discrepancy: velocity of sound from the gas composition minus the measured one
        vos_from_gas = df["vos_gc"]
        vos_measured = df["vos_avg"]
        df["gc_disc"] = vos_from_gas - vos_measured
        return df

    def load(self):
        """ Finds the right sheets, uses the other functions
        adds gass composition
        """
        excel_file = pd.ExcelFile(self.path)
        all_sheet_names = excel_file.sheet_names
        # Sheets with "run" in the name are the meter data
        meter_data_sheets = []
        for sheet_name in all_sheet_names:
            if "run" in sheet_name.lower():
                meter_data_sheets.append(sheet_name)

        # The sheet with "gas" in the name is the gas composition (take the first one)
        gas_sheets = []
        for sheet_name in all_sheet_names:
            if "gas" in sheet_name.lower():
                gas_sheets.append(sheet_name)
        gas_sheet = gas_sheets[0]


        self.gas = self.read_sheet(gas_sheet, GAS_RENAME)

        # a dictionary: a set of labelled tables
        self.meter_data = {}
        for sheet_name in meter_data_sheets:
            df = self.read_sheet(sheet_name, RENAME)


            # Attach the gas composition to each minute
            gas_without_vos = self.gas.drop(columns="vos_gc")
            df = df.join(gas_without_vos, how="left")
            # First draft of the zero-flow flag: True where flow is 0 (or missing)
            flow_with_gaps_filled = df["q_vol"].fillna(0)
            df["zero_flow"] = flow_with_gaps_filled.eq(0)
            self.meter_data[sheet_name] = self.add_indicators(df)

        return self.meter_data, self.gas


    def summary(self):
        """Print the size of every table."""
        for name, df in self.meter_data.items():
            number_of_zero_flow = int(df["zero_flow"].sum())
            number_of_missing = int(df.isna().sum().sum())
            print(name, df.shape, "zero-flow rows:", number_of_zero_flow, "NaNs:", number_of_missing)

            other_column_names.append(f"vos_{other}")

        # axis=1 means: average across the row
        average_of_others = other_paths_vos.mean(axis=1)

        df[f"dev_{path}"] = this_path_vos - average_of_others

    vos_from_gas = df["vos_gc"]
    vos_measured = df["vos_avg"]
    df["gc_disc"] = vos_from_gas - vos_measured
    return df

models ¶

Step 3 of RQ2: forecasting models with process inputs, compared against the baselines.

Two models, both fed exactly the same inputs (see forecast_data.py): lag_0 ... lag_{L-1}, seg_mean, since_start, q_vol, p, T, mw

LinearARX Ridge-regularised linear autoregression with exogenous inputs. A weighted sum of the inputs, so every weight can be read directly. The ridge strength is picked on the validation split.

XGBoostForecaster Gradient-boosted decision trees. Can learn non-linear effects and interactions (for example "low flow AND rising deviation"). The number of trees is picked on the validation split with early stopping.

Both predict the CHANGE from the last value (y - lag_0) and add it back, so a model that learns nothing useful falls back to persistence. The test split is only used for final scores.

ModelEvaluation reuses the datasets and splits from a BaselineEvaluation, so models and baselines are scored on exactly the same samples.

Usage

ev = BaselineEvaluation(p, horizons=(1, 5, 15), lookback=10).run() me = ModelEvaluation(ev).run().report() me.save("results")

class XGBoostForecaster(Model):
    name = "xgboost"

    def __init__(self, predict_delta=True, max_trees=1000, early_stopping=50, **params):
        """params are passed to XGBRegressor and override the defaults below."""
        self.predict_delta = predict_delta
        self.max_trees = max_trees
        self.early_stopping = early_stopping
        self.params = {
            "max_depth": 4,             # shallow trees: less overfitting on small data
            "learning_rate": 0.05,      # small steps, many trees
            "subsample": 0.8,           # each tree sees 80 % of the samples
            "colsample_bytree": 0.8,    # and 80 % of the inputs
            "min_child_weight": 5,      # no leaf for fewer than ~5 samples
            "random_state": 0,
            **params,
        }
        self.n_trees_ = None        # number of trees kept after early stopping
        self.importance_ = None     # share of the model's gain from each input (sums to 1)

    def _base(self, X):
        return X["lag_0"].to_numpy() if self.predict_delta else 0.0

    def fit(self, train, val=None):
        self.cols = list(train.X.columns)
        t = train.y.to_numpy() - self._base(train.X)

        self.model_ = XGBRegressor(
            n_estimators=self.max_trees,
            early_stopping_rounds=self.early_stopping if val is not None else None,
            eval_metric="mae",
            **self.params,
        )
        if val is not None:
            # Stop adding trees once validation MAE has not improved for `early_stopping` rounds.
            t_val = val.y.to_numpy() - self._base(val.X)
            self.model_.fit(train.X[self.cols], t, eval_set=[(val.X[self.cols], t_val)], verbose=False)
            self.n_trees_ = self.model_.best_iteration + 1
        else:
            self.model_.fit(train.X[self.cols], t, verbose=False)
            self.n_trees_ = self.max_trees

        self.importance_ = pd.Series(self.model_.feature_importances_, index=self.cols)
        return self

    def predict(self, X):
        return self._base(X) + self.model_.predict(X[self.cols])

class ModelEvaluation:
    def __init__(self, baseline_eval, make_models=default_models):
        self.ev = baseline_eval
        self.make_models = make_models
        self.results = pd.DataFrame()
        self.coefs = {}             # (target, horizon, model) -> signed weights (linear models)
        self.importances = {}       # (target, horizon, model) -> importance shares (tree models)

    def run(self):
        ev = self.ev
        if not ev.datasets:
            raise ValueError("run the BaselineEvaluation first")
        rows = []
        for (target, h), ds in ev.datasets.items():
            splits = ds.split(ev.train, ev.val)
            base = ev.results[(ev.results["target"] == target) & (ev.results["horizon"] == h)]

            # Honest comparison: choose the best baseline on VALIDATION, report its TEST score.
            val = base[base["split"] == "val"].set_index("baseline")["mae"]
            chosen = val.idxmin()
            test = base[base["split"] == "test"].set_index("baseline")["mae"]
            chosen_test = test[chosen]
            pers_test = test.get("persistence", np.nan)

            for m in self.make_models():
                m.fit(splits["train"], splits["val"])
                if getattr(m, "coef_", None) is not None:
                    self.coefs[(target, h, m.name)] = m.coef_
                if getattr(m, "importance_", None) is not None:
                    self.importances[(target, h, m.name)] = m.importance_

                for name in ("val", "test"):
                    s = splits[name]
                    pred = m.predict(s.X)
                    score = mae(s.y, pred)
                    row = {"target": target, "horizon": h, "split": name, "model": m.name,
                           "n": len(s), "mae": score, "rmse": rmse(s.y, pred),
                           "alpha": getattr(m, "alpha_", np.nan),
                           "n_trees": getattr(m, "n_trees_", np.nan)}
                    if name == "test":
                        row.update({"best_baseline": chosen, "baseline_mae": chosen_test,
                                    "skill_vs_persistence": 1 - score / pers_test,
                                    "skill_vs_best_baseline": 1 - score / chosen_test})
                    rows.append(row)
        self.results = pd.DataFrame(rows)
        return self

    def comparison(self):
        """Test MAE per model vs the best baseline (chosen on validation) vs persistence."""
        r = self.results[self.results["split"] == "test"]
        pers = self.ev.results[(self.ev.results["split"] == "test") &
                               (self.ev.results["baseline"] == "persistence")] \
            .set_index(["target", "horizon"])["mae"].rename("persistence_mae")
        out = r.set_index(["target", "horizon"])[["model", "mae", "best_baseline", "baseline_mae",
                                                  "skill_vs_persistence", "skill_vs_best_baseline"]]
        return out.join(pers)

    def top_coefficients(self, n=3):
        """Linear model: the inputs with the largest weight (per +1 std), with sign."""
        rows = []
        for (target, h, model), c in self.coefs.items():
            top = c.reindex(c.abs().sort_values(ascending=False).index)[:n]
            rows.append({"target": target, "horizon": h, "model": model,
                         "top_inputs": ", ".join(f"{k} ({v:+.4f})" for k, v in top.items())})
        return pd.DataFrame(rows).set_index(["target", "horizon", "model"])

    def top_importances(self, n=3):
        """Tree model: the inputs the trees used most (share of total gain, no sign)."""
        rows = []
        for (target, h, model), imp in self.importances.items():
            top = imp.sort_values(ascending=False)[:n]
            rows.append({"target": target, "horizon": h, "model": model,
                         "top_inputs": ", ".join(f"{k} ({v:.0%})" for k, v in top.items())})
        return pd.DataFrame(rows).set_index(["target", "horizon", "model"])

    def report(self):
        with pd.option_context("display.width", 220, "display.max_columns", 20,
                               "display.max_colwidth", 80, "display.float_format", "{:.4f}".format):
            print("\nTest MAE (m/s): models vs best baseline (picked on validation) vs persistence")
            print(self.comparison())
            if self.coefs:
                print("\nLinear ARX, largest weights per +1 std of the input:")
                print(self.top_coefficients())
            if self.importances:
                print("\nXGBoost, most used inputs (share of gain):")
                print(self.top_importances())
            print("\nskill_vs_best_baseline > 0 means the model beats the best baseline. "
                  "With few test samples, differences of a few percent are noise.")
        return self

    def save(self, out_dir):
        out = Path(out_dir)
        out.mkdir(parents=True, exist_ok=True)
        self.results.to_csv(out / "model_results.csv", index=False)
        self.comparison().to_csv(out / "model_vs_baselines.csv")
        if self.importances:
            pd.DataFrame(self.importances).T.to_csv(out / "xgboost_importances.csv")
        print(f"  saved results to {out}")
        return self

    """Makes the plots for each run DataFrame produced by USMLoader.

    Usage:
        runs, gas = USMLoader().load()
        plotter = USMPlotter(runs)

        self.out_dir = out_dir
        self.dpi = dpi
        # sheet name -> readable title; falls back to the sheet name if there are more runs than titles
        self.titles = {n: (RUN_TITLES[i] if i < len(RUN_TITLES) else n) for i, n in enumerate(runs)}

    def style(ax):

        for s in ("top", "right"):
            ax.spines[s].set_visible(False)
        ax.tick_params(labelsize=8, colors=MUTED)
        ax.xaxis.set_major_formatter(mdates.DateFormatter("%H:%M"))

    @staticmethod
    def plot_paths(d, ax, col):
        for p in PATHS:

        self.plot_paths(d, ax, "vos")
        ax.plot(d.index, d.vos_gc, color=INK, lw=1.2, ls="--", label="from GC")

    def plot_dev(self, d, ax):
        self.plot_paths(d, ax, "dev")

class Prepare:
    """Cleans one run: regular time grid, flags, `valid` mask and `segment` ids."""

    def __init__(self, df, name="meter_data", freq=FREQ, q_zero=Q_ZERO, snr_min=SNR_MIN,
                 vos_range=VOS_RANGE, max_ffill=MAX_FFILL_MIN, min_segment_len=MIN_SEGMENT_LEN):
        self.raw = df                     
        self.df = df.copy()
        self.name = name
        self.freq = freq
        self.q_zero = q_zero
        self.snr_min = snr_min
        self.vos_range = vos_range
        self.max_ffill = max_ffill
        self.min_segment_len = min_segment_len


    def add_indicators(self):
        """Recompute after reindexing so added rows are NaN, not stale."""
        self.df = USMLoader.add_indicators(self.df)
        return self

    def flag(self):
        """Per-row flags. Nothing is deleted.
            Checks where there is zero-flow,
            snr values are ok,
            vos values ok.
        """
        df = self.df
        df["zero_flow"] = df["q_vol"].abs().le(self.q_zero)

        snr_cols = [f"snr_{p}" for p in PATHS]
        vos_cols = [f"vos_{p}" for p in PATHS]
        # NaN counts as "not bad" here; NaN rows are caught by  has_inputs
        df["snr_ok"] = ~(df[snr_cols] <= self.snr_min).any(axis=1)
        df["vos_ok"] = ~((df[vos_cols] < self.vos_range[0]) |
                         (df[vos_cols] > self.vos_range[1])).any(axis=1)
        return self

    def mask(self):
        """`valid` = rows models may use; `segment` = id of each unbroken block of valid rows."""
        df = self.df
        has_inputs = df[INDICATORS + ["p", "T"] + self._composition_cols()].notna().all(axis=1)
        # Valid if flow is ok, no zero-flow 
        df["valid"] = (~df.zero_flow & df.snr_ok & df.vos_ok & has_inputs)

        # Step 2: find every run of consecutive valid rows
        # A run is stored as (start, end), where end is the first row AFTER the run.
        valid_flags = df["valid"].to_numpy()
        runs = []
        run_start = None                        

        for row_number, is_valid in enumerate(valid_flags):
            if is_valid and run_start is None:
                run_start = row_number           # a run begins here
            elif not is_valid and run_start is not None:
                runs.append((run_start, row_number))   # the run just ended
                run_start = None

        if run_start is not None:                # the data ended while still inside a run
            runs.append((run_start, len(valid_flags)))

        # Step 3: keep only long runs, and give each one a number 1, 2, 3, ...
        new_valid = np.zeros(len(df), dtype=bool)      # start with everything invalid
        new_segment = np.full(len(df), np.nan)         # start with everything NaN
        segment_number = 0

        for start, end in runs:
            run_length = end - start
            if run_length >= self.min_segment_len:
                segment_number += 1
                new_valid[start:end] = True
                new_segment[start:end] = segment_number

        df["valid"] = new_valid
        df["segment"] = new_segment
        return self

    def prepare(self):
        """Run all steps in order, starting from the raw data."""
        self.df = self.raw.copy()

        return (self.add_indicators()
                    .flag()
                    .mask())

    # ---- output ------------------------------------------------------------------
    @property
    def valid_df(self):
        """Only the rows models may use."""
        return self.df[self.df["valid"]]

    def report(self):
        "makes a report."
        df = self.df
        print(f"\n{self.name}: {len(df)} rows, {df.index.min()} -> {df.index.max()}")
        print(f"   low SNR: {int((~df.snr_ok).sum())}   implausible VOS: {int((~df.vos_ok).sum())}")

        n_seg = int(df.segment.max()) if df.valid.any() else 0
        print(f"  valid rows: {int(df.valid.sum())} ({df.valid.mean():.1%}) in {n_seg} segment(s)")
        if df.valid.any():
            lens = df[df.valid].groupby("segment").size()
            print(f"  segment length (min): min {lens.min()}, median {int(lens.median())}, max {lens.max()}")


        return self

    def save(self, out_dir):
        """saves the prepared csv. Input is the output directory."""
        out = Path(out_dir)
        out.mkdir(parents=True, exist_ok=True)
        fname = out / (self.name.replace(" ", "_") + "_prepared.csv")
        self.df.to_csv(fname)
        print(f"  saved {fname}")
        return self

    # ---- helpers -----------------------------------------------------------------
    def _composition_cols(self):
        """Returns the name of composition columns"""
        return [c for c in COMPOSITION if c in self.df.columns]

Input¶

A prepared run: either a Prepare object from prepare.py (anything with a .df attribute), or a DataFrame with a one-minute DatetimeIndex and these columns:

class StabilityAnalysis:
    def __init__(self, source, name=None, indicators=INDICATORS, process_vars=PROCESS_VARS,
                  rho_thresh=RHO_THRESHOLD, r2_thresh=R2_THRESHOLD):

        df = source.df
        self.name = name or source.name
        print("name",self.name)
        self.full = df
        self.indicators = [c for c in indicators if c in df.columns]
        self.rho_thresh = rho_thresh
        self.r2_thresh = r2_thresh
        self.trim_min = TRIM_MIN

        self.data = self._trimmed(df)
        wanted = [c for c in process_vars if c in df.columns]
        self.process_vars = [c for c in wanted if self.data[c].nunique() > 1 and self.data[c].std() > 0]
        self.constant_vars = [c for c in wanted if c not in self.process_vars]

    # ---- data ------------------------------------------------------------------------
    def _trimmed(self, df):
        v = df[df["valid"]]
        pos = v.groupby("segment").cumcount()
        return v[pos >= self.trim_min]

    # ---- 1. baseline -------------------------------------------------------------------
    def describe(self):
        rows = []
        for ind in self.indicators:
            s = self.data[ind].dropna()


            g = self.data.groupby("segment")[ind]


            n, var = g.count(), g.var(ddof=1)
            ok = n > 1
            within = np.sqrt((var[ok] * (n[ok] - 1)).sum() / (n[ok] - 1).sum()) if ok.any() else np.nan
            between = g.mean().std(ddof=1) if g.ngroups > 1 else np.nan
            days = (s.index - s.index[0]).total_seconds().to_numpy() / 86400 if len(s) else np.array([])
            drift = np.polyfit(days, s.to_numpy(), 1)[0] if len(s) > 2 and np.ptp(days) > 0 else np.nan
            rows.append({"indicator": ind, "n": len(s), "segments": g.ngroups,
                         "mean": s.mean(), "std": s.std(), "min": s.min(), "max": s.max(),
                         "std_within_segment": within, "std_between_segments": between,
                         "drift_per_day": drift})
        return pd.DataFrame(rows).set_index("indicator")

    # ---- 2. dependence on process conditions ---------------------------------------------
    def dependence(self):
        """Spearman correlation: rows = indicators, columns = process variables."""
        Spearman_correlation_table = pd.DataFrame(index=self.indicators, columns=self.process_vars, dtype=float)
        for ind in self.indicators:
            for var in self.process_vars:
                Spearman_correlation_table.loc[ind, var] = self.data[ind].corr(self.data[var], method="spearman")
        return Spearman_correlation_table

    def _ols(self, ind):
        cols = self.process_vars
        if not cols:
            return None
        d = self.data[[ind] + cols].dropna()
        if len(d) < len(cols) + 5:
            return None
        z = (d[cols] - d[cols].mean()) / d[cols].std()
        X = np.column_stack([np.ones(len(d)), z.to_numpy()])
        y = d[ind].to_numpy()
        beta = np.linalg.lstsq(X, y, rcond=None)[0]
        resid = y - X @ beta
        sst = ((y - y.mean()) ** 2).sum()
        r2 = 1 - (resid ** 2).sum() / sst if sst > 0 else np.nan
        resid_std = np.sqrt((resid ** 2).sum() / (len(d) - X.shape[1]))
        return dict(zip(cols, beta[1:])), r2, resid_std

    def effects(self):
        """Change in the indicator for +1 standard deviation of each process variable (all
        variables fitted together, so shared movement is partly separated)."""
        out = pd.DataFrame(index=self.indicators, columns=self.process_vars, dtype=float)
        for ind in self.indicators:
            res = self._ols(ind)
            if res:
                for var, b in res[0].items():
                    out.loc[ind, var] = b
        return out

    def fit(self):
        rows = {}
        for ind in self.indicators:
            res = self._ols(ind)
            rows[ind] = {"r2": res[1], "resid_std": res[2]} if res else {"r2": np.nan, "resid_std": np.nan}
        return pd.DataFrame(rows).T

    def collinearity(self, thresh=0.9):
        """Pairs of process variables that move together; their separate effects are unreliable."""
        c = self.data[self.process_vars].corr(method="spearman")
        pairs = [(a, b, c.loc[a, b]) for i, a in enumerate(c.columns) for b in c.columns[i + 1:]
                 if abs(c.loc[a, b]) >= thresh]
        return pd.DataFrame(pairs, columns=["var_a", "var_b", "rho"])

    # ---- 3. composition steps -------------------------------------------------------------
    def step_response(self, window=5, mw_tol=MW_TOL):
        """Mean indicator after minus before each composition update, inside one segment."""
        df = self.full
        if "mw" not in df.columns:
            return pd.DataFrame(columns=["time", "d_mw", "indicator", "delta"])
        events = df.index[df["mw"].diff().abs() > mw_tol]
        v = df[df["valid"]]
        w, one = pd.Timedelta(minutes=window), pd.Timedelta(minutes=1)
        rows = []
        for t in events:
            if not df.loc[t, "valid"]:
                continue
            s = v[v["segment"] == df.loc[t, "segment"]]
            before, after = s.loc[t - w:t - one], s.loc[t:t + w - one]
            if len(before) < 2 or len(after) < 2:
                continue
            for ind in self.indicators:
                rows.append({"time": t, "d_mw": df["mw"].diff().loc[t], "indicator": ind,
                             "delta": after[ind].mean() - before[ind].mean()})
        return pd.DataFrame(rows, columns=["time", "d_mw", "indicator", "delta"])

    def step_summary(self, **kw):
        st = self.step_response(**kw)
        if st.empty:
            return pd.DataFrame(columns=["events", "median_abs_delta", "max_abs_delta"])
        g = st.assign(a=st["delta"].abs()).groupby("indicator")["a"]
        return pd.DataFrame({"events": g.count(), "median_abs_delta": g.median(),
                             "max_abs_delta": g.max()})

    # ---- 4. verdict -----------------------------------------------------------------------
    def summary(self):
        desc, rho, fit = self.describe(), self.dependence(), self.fit()
        rows = []
        for ind in self.indicators:
            depends = [v for v in self.process_vars if abs(rho.loc[ind, v]) >= self.rho_thresh]
            r2 = fit.loc[ind, "r2"]
            if depends or (pd.notna(r2) and r2 >= self.r2_thresh):
                verdict = "process-dependent"
            else:
                verdict = "stable"
            row = {"indicator": ind, "std": desc.loc[ind, "std"],
                   "std_within_segment": desc.loc[ind, "std_within_segment"],
                   "drift_per_day": desc.loc[ind, "drift_per_day"],
                   "r2": r2, "unexplained_std": fit.loc[ind, "resid_std"],
                   "depends_on": ", ".join(depends) or "-", "verdict": verdict}
            rows.append(row)
        return pd.DataFrame(rows).set_index("indicator")

    # ---- output ---------------------------------------------------------------------------
    def report(self):
        pd.set_option("display.width", 200, "display.max_columns", 20, "display.float_format", "{:.4f}".format)
        print(f"\n=== Stability: {self.name} ===")
        print(f"valid rows used: {len(self.data)} (first {self.trim_min} min of each segment dropped), "
              f"segments: {self.data['segment'].nunique()}")
        if self.constant_vars:
            print(f"  constant in this data, so their effect cannot be assessed: {self.constant_vars}")
        print("\nBaseline:\n", self.describe())
        print("\nSpearman correlation with process variables:\n", self.dependence())
        print("\nEffect of +1 std of each variable (indicator units):\n", self.effects())
        col = self.collinearity()
        if not col.empty:
            print("\nWarning: these variables move together, separate effects are unreliable:\n", col)
        st = self.step_summary()
        if not st.empty:
            print("\nChange across composition updates:\n", st)
        print("\nSummary (heuristic verdict):\n", self.summary())
        return self

    def plot(self, path):

        cols = self.process_vars
        fig, axes = plt.subplots(len(self.indicators), 1 + len(cols),
                                 figsize=(3.4 * (1 + len(cols)), 2.4 * len(self.indicators)),
                                 squeeze=False, constrained_layout=True)
        rho = self.dependence()
        ink, muted, blue = "#0b0b0b", "#52514e", "#2a78d6"
        for i, ind in enumerate(self.indicators):

    for ind in self.indicators:
        for var in self.process_vars:
            Spearman_correlation_table.loc[ind, var] = self.data[ind].corr(self.data[var], method="spearman")

def collinearity(self, thresh=0.9):
    """Pairs of process variables that move together; their separate effects are unreliable."""
    c = self.data[self.process_vars].corr(method="spearman")
    pairs = [(a, b, c.loc[a, b]) for i, a in enumerate(c.columns) for b in c.columns[i + 1:]
             if abs(c.loc[a, b]) >= thresh]
    return pd.DataFrame(pairs, columns=["var_a", "var_b", "rho"])

def step_response(self, window=5, mw_tol=MW_TOL):
    """Mean indicator after minus before each composition update, inside one segment."""
    df = self.full
    if "mw" not in df.columns:
        return pd.DataFrame(columns=["time", "d_mw", "indicator", "delta"])
    events = df.index[df["mw"].diff().abs() > mw_tol]
    v = df[df["valid"]]
    w, one = pd.Timedelta(minutes=window), pd.Timedelta(minutes=1)
    rows = []
    for t in events:
        if not df.loc[t, "valid"]:
            continue
        s = v[v["segment"] == df.loc[t, "segment"]]
        before, after = s.loc[t - w:t - one], s.loc[t:t + w - one]
        if len(before) < 2 or len(after) < 2:
            continue
        for ind in self.indicators:
            rows.append({"time": t, "d_mw": df["mw"].diff().loc[t], "indicator": ind,
                         "delta": after[ind].mean() - before[ind].mean()})
    return pd.DataFrame(rows, columns=["time", "d_mw", "indicator", "delta"])
